```typescript
export const performanceAPI = {
  getMetrics: (athleteId: string) => api.get(`/api/performance/metrics/${athleteId}`),
  getTrends: (athleteId: string, metric: string, days = 90) =>
    api.get(`/api/performance/trends/${athleteId}?metric=${metric}&days=${days}`),
  getTrainingLoad: (athleteId: string, days = 90) =>
    api.get(`/api/performance/load/${athleteId}?days=${days}`),
  getComparison: (athleteId: string) => api.get(`/api/performance/comparison/${athleteId}`),
};
```

In `frontend/src/hooks/usePerformance.ts`, change ALL hook parameter types from `number` to `string` and add `enabled: !!athleteId` guards:

```typescript
export function usePerformanceMetrics(athleteId: string) {
  return useQuery({
    queryKey: ["performance", "metrics", athleteId],
    queryFn: () => performanceAPI.getMetrics(athleteId).then((res) => res.data),
    enabled: !!athleteId,
    staleTime: 60 * 60 * 1000,
  });
}

export function usePerformanceTrends(athleteId: string, metric: string, days = 90) {
  return useQuery({
    queryKey: ["performance", "trends", athleteId, metric, days],
    queryFn: () => performanceAPI.getTrends(athleteId, metric, days).then((res) => res.data),
    enabled: !!athleteId && !!metric,
    staleTime: 60 * 60 * 1000,
  });
}

export function useTrainingLoad(athleteId: string, days = 90) {
  return useQuery({
    queryKey: ["performance", "load", athleteId, days],
    queryFn: () => performanceAPI.getTrainingLoad(athleteId, days).then((res) => res.data),
    enabled: !!athleteId,
    staleTime: 5 * 60 * 1000, // 5 min for PMC views (shorter than metrics)
  });
}

export function usePeerComparison(athleteId: string) {
  return useQuery({
    queryKey: ["performance", "comparison", athleteId],
    queryFn: () => performanceAPI.getComparison(athleteId).then((res) => res.data),
    enabled: !!athleteId,
    staleTime: 24 * 60 * 60 * 1000,
  });
}
```

Do NOT change any other API sections in api.ts (trainingAPI, wellbeingAPI, etc.).
