Add a new endpoint in `backend/app/api/endpoints/training.py`:

```python
@router.get("/coach/athletes/{athlete_id}/workouts")
async def get_athlete_workouts_for_coach(
    athlete_id: str,
    limit: int = Query(20),
    date_from: Optional[str] = Query(None),
    date_to: Optional[str] = Query(None),
    user=Depends(get_current_user),
):
    """Get an athlete's workouts for coach review. Requires coach role."""
    try:
        # Check if user has coach role
        role = user.user_metadata.get("role", "athlete") if user.user_metadata else "athlete"
        # Also check app_metadata (role may be stored there per Phase 2 decision)
        if not role or role == "athlete":
            role = user.app_metadata.get("role", "athlete") if user.app_metadata else "athlete"
        if role != "coach":
            raise HTTPException(status_code=403, detail="Coach role required")

        supabase = get_supabase()
        query = (
            supabase.table("workouts")
            .select("*, workout_exercises(*, exercises(name, category))")
            .eq("athlete_id", athlete_id)
            .order("date", desc=True)
            .limit(limit)
        )
        if date_from:
            query = query.gte("date", date_from)
        if date_to:
            query = query.lte("date", date_to)

        result = query.execute()
        return result.data
    except HTTPException:
        raise
    except RuntimeError as e:
        raise HTTPException(status_code=503, detail=str(e))
    except Exception as e:
        raise HTTPException(
            status_code=500, detail=f"Failed to fetch athlete workouts: {str(e)}"
        )
```

```python
@router.get("/coach/athletes/workout-status")
async def get_athletes_workout_status(
    days: int = Query(7),
    user=Depends(get_current_user),
):
    """Get workout completion status for all athletes (coach only).
    Returns each athlete's name, total workouts, completed count for the period."""
    try:
        role = user.user_metadata.get("role", "athlete") if user.user_metadata else "athlete"
        if not role or role == "athlete":
            role = user.app_metadata.get("role", "athlete") if user.app_metadata else "athlete"
        if role != "coach":
            raise HTTPException(status_code=403, detail="Coach role required")

        supabase = get_supabase()
        from datetime import date, timedelta
        date_from = (date.today() - timedelta(days=days)).isoformat()

        # Get all athletes (simplified — in production, filter by coach-athlete relationship)
        athletes_result = supabase.table("athletes").select("id, first_name, last_name").eq("is_active", True).execute()

        statuses = []
        for athlete in athletes_result.data:
            workouts_result = (
                supabase.table("workouts")
                .select("id, is_completed, date, name")
                .eq("athlete_id", athlete["id"])
                .gte("date", date_from)
                .execute()
            )
            workouts = workouts_result.data
            statuses.append({
                "athlete_id": athlete["id"],
                "athlete_name": f"{athlete['first_name']} {athlete['last_name']}",
                "total_workouts": len(workouts),
                "completed": sum(1 for w in workouts if w.get("is_completed")),
                "recent_workouts": workouts[:5],
            })

        return statuses
    except HTTPException:
        raise
    except RuntimeError as e:
        raise HTTPException(status_code=503, detail=str(e))
    except Exception as e:
        raise HTTPException(
            status_code=500, detail=f"Failed to fetch workout status: {str(e)}"
        )
```
