In `frontend/src/hooks/useTraining.ts`, update `useWorkouts` to accept filter params:
```typescript
interface WorkoutFilters {
  workout_type?: string;
  date_from?: string;
  date_to?: string;
  search?: string;
  limit?: number;
  offset?: number;
}

export function useWorkouts(filters: WorkoutFilters = {}) {
  return useQuery({
    queryKey: ["workouts", filters],
    queryFn: () => {
      const params = new URLSearchParams();
      Object.entries(filters).forEach(([k, v]) => {
        if (v !== undefined && v !== '') params.set(k, String(v));
      });
      return api.get(`/api/training/workouts?${params.toString()}`).then((res) => res.data);
    },
    staleTime: 2 * 60 * 1000,
  });
}
```

Add `useWeeklyWorkouts` hook:
```typescript
export function useWeeklyWorkouts(weekStart: string) {
  return useQuery({
    queryKey: ["workouts", "week", weekStart],
    queryFn: () => api.get(`/api/training/workouts/week?week_start=${weekStart}`).then((res) => res.data),
    staleTime: 2 * 60 * 1000,
    enabled: !!weekStart,
  });
}
```
